==================
Default prefixes retain complete clause conditions
==================
switch (1) {
  default"x" {}
  default'name' {}
  default$value {}
  default${name} {}
  default$(Get-Item .) {}
  default {}
}

==================
A newline may precede the first clause and a semicolon may follow it
==================
switch (1) {
  default # fallback
  {} ;
}

==================
A semicolon before the first clause is an error
==================
switch (1) { ; default {} }

==================
A file switch also rejects a leading semicolon
==================
switch -file input.txt {
; default {} }
